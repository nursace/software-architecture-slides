% --- Encoding & language (pdflatex + Cyrillic) ---
% Last encoding in \usepackage[...]{fontenc} becomes the document default.

% listings: generic / light code snippets without shell-escape
% Keep Latin-looking listings; body text uses T2A via babel/russian.
\lstdefinestyle{codestyle}{
  basicstyle=\ttfamily\small,
  keywordstyle=\color{KeywordBlue}\bfseries,
  commentstyle=\color{CommentGray}\itshape,
  stringstyle=\color{StringGreen},
  numbers=left,
  numberstyle=\tiny\color{CommentGray},
  numbersep=6pt,
  frame=single,
  rulecolor=\color{LectureGray!40},
  backgroundcolor=\color{CodeBg},
  breaklines=true,
  showstringspaces=false,
  tabsize=4,
  captionpos=b,
}
\lstset{style=codestyle}

\newcommand{\quizpause}{%

}

  \end{block}%

  \setbeamercolor{block body}{bg=CodeBg,fg=black}%
}
